// 1. RenderContext Definition
interface RenderContextType {
  theme: string;
  motion: string;
  assets: Record<string, any>;
  invitationData: any;
}

const RenderContext = createContext<RenderContextType | null>(null);

export function useRenderContext() {
  const ctx = useContext(RenderContext);
  if (!ctx) {
    throw new Error("useRenderContext must be used within a RenderProvider");
  }
  return ctx;
}

// 2. Profiling Hook Implementation
export interface ProfileMetrics {
  renderTimeMs: number;
  sectionsRendered: string[];
  skippedSections: string[];
  failedSections: string[];
}

export function useRendererProfiler(onProfileComplete?: (metrics: ProfileMetrics) => void) {
  const startTime = useRef(0);
  const metrics = useRef<ProfileMetrics>({
    renderTimeMs: 0,
    sectionsRendered: [],
    skippedSections: [],
    failedSections: []
  });

  const startProfiling = () => {
    startTime.current = performance.now();
    metrics.current = {
      renderTimeMs: 0,
      sectionsRendered: [],
      skippedSections: [],
      failedSections: []
    };
  };

  const recordRender = (id: string) => {
    if (!metrics.current.sectionsRendered.includes(id)) {
      metrics.current.sectionsRendered.push(id);
    }
  };

  const recordSkip = (id: string) => {
    if (!metrics.current.skippedSections.includes(id)) {
      metrics.current.skippedSections.push(id);
    }
  };

  const recordFailure = (id: string) => {
    if (!metrics.current.failedSections.includes(id)) {
      metrics.current.failedSections.push(id);
    }
  };

  const stopProfiling = () => {
    if (startTime.current > 0) {
      metrics.current.renderTimeMs = performance.now() - startTime.current;
      if (onProfileComplete) {
        onProfileComplete(metrics.current);
      }
    }
    return metrics.current;
  };

  return { startProfiling, recordRender, recordSkip, recordFailure, stopProfiling, metrics: metrics.current };
}
