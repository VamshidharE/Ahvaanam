 	"use client";

import React, { useRef, useState } from "react";
import { Volume2, VolumeX } from "lucide-react";

interface AudioOrchestratorProps {
  musicFile?: string;
  musicEnabled?: boolean;
}

export default function AudioOrchestrator({ musicFile, musicEnabled }: AudioOrchestratorProps) {
  const [playingMusic, setPlayingMusic] = useState(false);
  const audioRef = useRef<HTMLAudioElement | null>(null);

  if (!musicEnabled || !musicFile) return null;

  const toggleMusic = () => {
    if (!audioRef.current) return;
    if (playingMusic) {
      audioRef.current.pause();
    } else {
      audioRef.current.play().catch(() => {});
    }
    setPlayingMusic(!playingMusic);
  };

  return (
    <>
      <audio ref={audioRef} loop src={musicFile} />
      <button
        onClick={toggleMusic}
        className="fixed bottom-6 right-6 z-40 flex h-12 w-12 items-center justify-center rounded-full border border-orange-300/40 bg-white/90 text-[#FF8800] backdrop-blur-md shadow-md hover:scale-105 active:scale-95 transition-all cursor-pointer select-none"
      >
        {playingMusic ? <Volume2 size={20} /> : <VolumeX size={20} />}
      </button>
    </>
  );
}
