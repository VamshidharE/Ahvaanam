use client";

import React, { useState, useEffect } from "react";
import { useInvitationData } from "../../context/InvitationDataContext";

import AudioOrchestrator from "./AudioOrchestrator";
import BackgroundEffects from "./BackgroundEffects";
import LoadingExperience from "./LoadingExperience";
import SectionPipeline from "./SectionPipeline";
import TemplateFrame from "./Template    Frame";
import ScrollController from "./ScrollController";
import ShareController from "./ShareController";

interface LegacyOrchestratorProps {
  themeId: string;
  onBackToHub: () => void;
  particlesComponent?: React.ComponentType<any>;
  themeBgClass?: string;
  loadingTitle?: string;
  loadingSubtitle?: string;
}

export default function LegacyOrchestrator({
  themeId,
  onBackToHub,
  particlesComponent,
  themeBgClass = "bg-stone-900",
  loadingTitle = "Invitation",
  loadingSubtitle = "Celebrating Love"
}: LegacyOrchestratorProps) {
  const { invitationData } = useInvitationData();
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    if (invitationData) {
      const timer = setTimeout(() => setLoading(false), 1200);
      return () => clearTimeout(timer);
    }
  }, [invitationData]);

  if (!invitationData) return null;

  const isWrapped = invitationData && "category" in invitationData && "config" in invitationData;
  const invitation = isWrapped ? (invitationData as any).config : invitationData;
  const sectionOrder = invitation.sectionOrder || [];

  return (
    <ScrollController>
      <TemplateFrame themeBgClass={themeBgClass} onBackToHub={onBackToHub}>
        {/* Loading overlay screen */}
        <LoadingExperience loading={loading} title={loadingTitle} subtitle={loadingSubtitle} />

        {/* Particles canvas */}
        <BackgroundEffects
          showParticles={invitation.themeSettings?.showParticles}
          particlesComponent={particlesComponent}
        />

        {/* Audio background playback */}
        <AudioOrchestrator
          musicFile={invitation.music?.file}
          musicEnabled={invitation.music?.enabled}
        />

        {/* Share buttons */}
        <ShareController />

        {/* Sections pipeline render block */}
        <SectionPipeline
          sectionOrder={sectionOrder}
          invitation={invitation}
          themeId={themeId}
        />
      </TemplateFrame>
    </ScrollController>
  );
}
export { default as AudioOrchestrator } from "./AudioOrchestrator";
export { default as BackgroundEffects } from "./BackgroundEffects";
export { default as LoadingExperience } from "./LoadingExperience";
export { default as SectionPipeline } from "./SectionPipeline";
export { default as TemplateFrame } from "./TemplateFrame";
export { default as ScrollController } from "./ScrollController";
export { default as ShareController } from "./ShareController";

## 1. Tech Stack

### Core Frameworks & Runtime
* **Framework**: Next.js 16.2.9 (using App Router, Route Handlers, and Turbopack compiler wrapper).
* **Language**: TypeScript 5.x.
* **Component Engine**: React 19.2.4 (utilizing Server Components by default, client hooks, and lazy loading).

### Libraries & Utilities
* **Motion & Physics Engine**: Framer Motion 12.4.2 (handles reveals, hovers, drags, and spring physics), GSAP 3.15.0, and React Scroll Parallax 3.5.0.
* **Validation**: Zod 4.4.3 (rules engine validating section manifests).
* **Icons**: Lucide React 1.21.0, React Icons 5.6.0.
* **Lightbox**: Yet Another React Lightbox 3.32.0.

### State Management
* **Client State**: Native React Context API. The central coordinator is the `InvitationDataProvider` (context provider), storing local state and exposing methods to update fields and persist changes to Supabase.
* **Iframe Communication**: HTML5 Web postMessage API. Relays configuration objects in real-time between the parent editor workspace and the child preview iframe.

### Authentication, Database & ORM
* **Auth**: Supabase Auth (handled via `@supabase/ssr` cookies management).
* **Database**: Supabase PostgreSQL.
* **ORM**: None (raw Supabase JS Client `createClient` calling REST endpoints).
* **Storage**: Supabase Storage Buckets (handles media library uploads).

### Styling
* **CSS Framework**: Tailwind CSS v4.0.0 (utilizes PostCSS compiling utility rules).
* **Custom Themes**: Dynamic CSS Variables injected via a custom React `<ThemeProvider>` component.

## 3. Editor Architecture

The customizer is designed around a **Unidirectional Data Flow** with a nested iframe serving as an isolated preview sandbox.

### Data & State Flow Map

```mermaid
sequenceDiagram
    autonumber
    actor User as Designer/User
    participant WS as Editor Workspace (Admin Portal)
    participant DataCtx as InvitationDataContext
    participant Preview as Live Preview (Iframe sandbox)
    participant API as Supabase REST API Route
    participant DB as Postgres Table

    User->>WS: Edits Title / Swaps theme / Drags order
    WS->>DataCtx: updateInvitationData(updater)
    Note over DataCtx: React State updates locally
    DataCtx->>Preview: postMessage({ type: "UPDATE_CONFIG", config })
    Note over Preview: Iframe catches postMessage<br/>Triggers local React render
    Preview-->>User: Screen updates instantly (no refresh!)
    User->>WS: Clicks "Save / Publish"
    WS->>DataCtx: saveConfig()
    DataCtx->>API: POST /api/invitation/[id]
    API->>DB: UPDATE invitations SET config = json
    DB-->>User: Confetti chimes play!
```
