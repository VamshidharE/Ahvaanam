const fs = require('fs');
const path = require('path');

const brainDir = 'C:\\Users\\Admin\\.gemini\\antigravity\\brain';
const folders = fs.readdirSync(brainDir);

for (const folder of folders) {
  const folderPath = path.join(brainDir, folder);
  if (!fs.statSync(folderPath).isDirectory()) continue;
  
  const logPath = path.join(folderPath, '.system_generated', 'logs', 'transcript_full.jsonl');
  if (!fs.existsSync(logPath)) continue;
  
  try {
    const content = fs.readFileSync(logPath, 'utf8');
    const lines = content.split('\n').filter(l => l.trim());
    console.log(`Searching folder ${folder}: ${lines.length} lines`);
    
    let matchCount = 0;
    for (let i = 0; i < lines.length; i++) {
      const line = lines[i];
      try {
        const obj = JSON.parse(line);
        if (obj.tool_calls) {
          for (const tc of obj.tool_calls) {
            if (tc.args && tc.args.TargetFile && (tc.args.TargetFile.includes('composition') || tc.args.TargetFile.includes('CompositionRenderer'))) {
              matchCount++;
              console.log(`  [${matchCount}] Match in step ${obj.step_index}: tool=${tc.name}, TargetFile=${tc.args.TargetFile}`);
              
              let code = tc.args.CodeContent || tc.args.ReplacementContent;
              if (tc.name === 'multi_replace_file_content' && tc.args.ReplacementChunks) {
                code = JSON.stringify(tc.args.ReplacementChunks);
              }
              
              if (code) {
                const outName = `scratch/restored_${folder}_step_${obj.step_index}_${path.basename(tc.args.TargetFile)}.txt`;
                fs.writeFileSync(outName, code);
                console.log(`    Saved code to ${outName} (size: ${code.length})`);
              }
            }
          }
        }
      } catch (e) {}
    }
  } catch (err) {
    console.error(`Error in folder ${folder}:`, err.message);
  }
}
console.log('Composition code search complete.');
