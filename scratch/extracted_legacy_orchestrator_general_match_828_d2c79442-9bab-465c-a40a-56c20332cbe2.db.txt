      case "houseDetails":

        return null;

  return (
    <div className={`min-h-screen relative overflow-hidden ${themeBgClass}`}>
      {invitation.music?.enabled && invitation.music?.file && (
        <audio ref={audioRef} loop src={invitation.music.file} />
      )}

      {/* Loading overlay */}
      <AnimatePresence>
        {loading && (
          <motion.div
            initial={{ opacity: 1 }}
            exit={{ opacity: 0 }}
            className="fixed inset-0 z-50 bg-[#FFFDF5] flex flex-col items-center justify-center text-center p-6"
          >
            <h2 className="text-[#FF8800] text-xs uppercase tracking-[4px] mb-1 font-bold">{loadingTitle}</h2>
            <p className="text-amber-900/40 text-xs font-semibold">{loadingSubtitle}</p>
          </motion.div>
        )}
      </AnimatePresence>

      {/* Particles canvas overlay */}
      {invitation.themeSettings?.showParticles && Particles && <Particles />}

      {/* Floating utility buttons */}
      <div className="fixed bottom-6 right-6 z-40 flex flex-col gap-4">
        <button
          onClick={onBackToHub}
          className="flex h-12 w-12 items-center justify-center rounded-full border border-orange-300/40 bg-white/90 text-[#FF8800] backdrop-blur-md shadow-md hover:scale-105 active:scale-95 transition-all cursor-pointer"
        >
          <ArrowLeft size={20} />
        </button>
        {invitation.music?.enabled && (
          <button
            onClick={toggleMusic}
            className="flex h-12 w-12 items-center justify-center rounded-full border border-orange-300/40 bg-white/90 text-[#FF8800] backdrop-blur-md shadow-md hover:scale-105 active:scale-95 transition-all cursor-pointer"
          >
            {playingMusic ? <Volume2 size={20} /> : <VolumeX size={20} />}
          </button>
        )}
      </div>

      {/* Dynamic sections container */}
      <div className="space-y-12">
        {sectionOrder.map((sectionId: string) => renderSectionComponent(sectionId))}
      </div>
    </div>
  );
