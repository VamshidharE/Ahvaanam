omponentType<any>;
  themeBgClass?: string;
  loadingTitle?: string;
  loadingSubtitle?: string;
}

export default function LegacyOrchestrator({
  themeId,
  onBackToHub,
  particlesComponent: Particles,
  themeBgClass = "bg-stone-900",
  loadingTitle = "Invitation",
  loadingSubtitle = "Celebrating Love"
}: LegacyOrchestratorProps) {
  const { invitationData } = useInvitationData();
  const [loading, setLoading] = useState(true);
  const [playingMusic, setPlayingMusic] = useState(false);
  const audioRef = useRef<HTMLAudioElement | null>(null);

  useEffect(() => {
    if (invitationData) {
      const timer = setTimeout(() => setLoading(false), 1500);
      return () => clearTimeout(timer);
    }
  }, [invitationData]);

  if (!invitationData) return null;

  const isWrapped = invitationData && "category" in invitationData && "config" in invitationData;
  const invitation = isWrapped ? (invitationData as any).config : invitationData;

  const toggleMusic = () => {
    if (!audioRef.current) return;
    if (playingMusic) {
      audioRef.current.pause();
    } else {
      audioRef.current.play().catch(() => {});
    }
    setPlayingMusic(!playingMusic);
  };

  const sectionOrder = invitation.sectionOrder || [];

  const renderSectionCompo    nent = (sectionId: string) => {
    const sectionConfig = invitation[sectionId];
    const isEnabled = sectionConfig?.enabled !== false;
    if (!isEnabled) return null;

    const properties = sectionConfig || {};
    const bindings = sectionConfig || {};

    switch (sectionId) {
      case "hero":
        return (
          <HeroSection
            key={sectionId}
            properties={properties}
            bindings={bindings}
            themeId={themeId}
            layoutId="hero-center"
            motionId="luxury-motion"
          />
        );
      case "couple":
        return <CoupleSection key={sectionId} bindings={bindings} layoutId="hero-center" motionId="luxury-motion" />;
      case "countdown":
        return <CountdownSection key={sectionId} bindings={bindings} />;
      case "story":
        return <StorySection key={sectionId} bindings={bindings} />;
      case "events":
        return <EventsSection key={sectionId} bindings={bindings} />;
      case "gallery":
        return <GallerySection key={sectionId} bindings={bindings} />;
      case "venue":
        return <VenueSection key={sectionId} bindings={bindings} />;
      case "families":
        return <FamiliesSection key={sectionId} bindings={bindings} />;
      case "rsvp":
        return <RSVPSection key={sectionId} bindings={bindings} />;
      case "faq":
        return <FAQSection key={sectionId} bindings={bindings} />;
      case "dressCode":
        return <DressCodeSection key={sectionId} bindings={bindings} />;
      case "footer":
        return <FooterSection key={sectionId} bindings={bindings} />;
      case "celebrant":
        return <CelebrantSection key={sectionId} bindings={bindings} layoutId="hero-center" motionId="luxury-motion" />;
      case "houseDetails":
        return <HouseDetailsSection key={sectionId} bindings={bindings} />;
      case "owners":
        return <OwnersSection key={sectionId} bindings={bindings} />;
      default:
        return null;
    }
  };

  return (
    <div className={`min-h-screen relative overflow-hidden ${themeBgClass}`}>
      {invitation.music?.enabled && invitation.music?.file && (
        <audio ref={audioRef} loop src={invitation.music.file} />
      )}

      {/* Loading overlay */}
      <AnimatePresence>
        {loading && (
          <motion.div
            initial={{ opacity: 1 }}
            exit={{ opacity: 0 }}
            className="fixed inset-0 z-50 bg-[#FFFDF5] flex flex-col items-center justify-center text-center p-6"
          >
            <h2 className="text-[#FF8800] text-xs uppercase tracking-[4px] mb-1 font-bold">{loadingTitle}</h2>
            <p className="text-amber-900/40 text-xs font-semibold">{loadingSubtitle}</p>
          </motion.div>
        )}
      </AnimatePresence>

      {/* Particles canvas overlay */}
      {invitation.themeSettings?.showParticles && Particles && <Particles />}

      {/* Floating utility buttons */}
      <div className="fixed bottom-6 right-6 z-40 flex flex-col gap-4">
        <button
          onClick={onBackToHub}
          className="flex h-12 w-12 items-center justify-center rounded-full border border-orange-300/40 bg-white/90 text-[#FF8800] backdrop-blur-md shadow-md hover:scale-105 active:scale-95 transition-all cursor-pointer"
        >
          <ArrowLeft size={20} />
        </button>
        {invitation.music?.enabled && (
          <button
            onClick={toggleMusic}
            className="flex h-12 w-12 items-center justify-center rounded-full border border-orange-300/40 bg-white/90 text-[#FF8800] backdrop-blur-md shadow-md hover:scale-105 active:scale-95 transition-all cursor-pointer"
          >
            {playingMusic ? <Volume2 size={20} /> : <VolumeX size={20} />}
          </button>
        )}
      </div>

      {/* Dynamic sections container */}
      <div className="space-y-12">
        {sectionOrder.map((sectionId: string) => renderSectionComponent(sectionId))}
      </div>
    </div>
  );
}
