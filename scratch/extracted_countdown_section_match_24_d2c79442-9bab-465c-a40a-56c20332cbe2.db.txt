import React, { useEffect, useState } from "react";

interface CountdownSectionProps {
  bindings: {
    title?: string;
    targetDate?: string;
    postWeddingMessage?: string;
  };
}

export default function CountdownSection({ bindings }: CountdownSectionProps) {
  const { 
    title = "Party Countdown", 
    targetDate = "2026-12-14T19:00:00", 
    postWeddingMessage = "The Celebration has begun!" 
  } = bindings;

  const [timeLeft, setTimeLeft] = useState({ days: 0, hours: 0, minutes: 0, seconds: 0 });
  const [isPassed, setIsPassed] = useState(false);

  useEffect(() => {
    if (!targetDate) return;
    const targetTime = new Date(targetDate).getTime();

    const updateCountdown = () => {
      const now = new Date().getTime();
      const difference = targetTime - now;
      if (difference <= 0) {
        setIsPassed(true);
      } else {
        setIsPassed(false);
        setTimeLeft({
          days: Math.floor(difference / (1000 * 60 * 60 * 24)),
          hours: Math.floor((difference % (1000 * 60 * 60 * 24)) / (1000 * 60 * 60)),
          minutes: Math.floor((difference % (1000 * 60 * 60)) / (1000 * 60)),
          seconds: Math.floor((difference % (1000 * 60)) / 1000),
        });
      }
    };

    updateCountdown();
    const interval = setInterval(updateCountdown, 1000);
    return () => clearInterval(interval);
  }, [targetDate]);

  return (
    <section 
      className="py-20 px-6 border-b border-white/5 text-center select-none"
      style={{ 
        fontFamily: "var(--theme-font-body)",
        color: "var(--theme-text)",
        background: "rgba(139, 92, 246, 0.03)"
      }}
    >
      <div className="max-w-4xl mx-auto space-y-10">
        <div className="space-y-2">
          <h3 
            className="text-xs uppercase tracking-[3px] font-bold"
            style={{ 
              color: "var(--theme-accent)",
              fontFamily: "var(--theme-font-heading)" 
            }}
          >
            {title}
          </h3>
        </div>
        {isPassed ? (
          <p 
            className="text-lg uppercase tracking-[1.5px] font-bold animate-pulse"
            style={{ color: "var(--theme-accent)" }}
          >
            {postWeddingMessage}
          </p>
        ) : (
          <div className="grid grid-cols-4 gap-4 max-w-md mx-auto">
            {Object.entries(timeLeft).map(([label, value]) => (
              <div 
                key={label} 
                className="border border-white/5 py-4 flex flex-col justify-center shadow-xs"
                style={{ 
                  backgroundColor: "var(--theme-card-bg)",
                  borderColor: "var(--theme-border)",
                  borderRadius: "var(--theme-radius-card)",
                  boxShadow: "var(--theme-shadow-card)"
                }}
              >
                <span 
                  className="text-2xl md:text-3xl font-extrabold"
                  style={{ color: "var(--theme-accent)" }}
                >
                  {value}

                <span 
                  className="text-[9px] uppercase tracking-[1px] font-bold mt-1"
                  style={{ color: "var(--theme-secondary-text)" }}
                >
                  {label}

              </div>
            ))}
          </div>
        )}
      </div>
    </section>
  );
