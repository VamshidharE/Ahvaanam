"use client";

import React, { useState, useEffect } from "react";
import { useInvitationData } from "../context/InvitationDataContext";

import AudioOrchestrator from "./AudioOrchestrator";
import BackgroundEffects from "./BackgroundEffects";
import LoadingExperience from "./LoadingExperience";
import SectionPipeline from "./SectionPipeline";
import TemplateFrame from "./TemplateFrame";
import ScrollController from "./ScrollController";
import ShareController from "./ShareController";

interface LegacyOrchestratorProps {
  themeId: string;
  onBackToHub: () => void;
  particlesComponent?: React.ComponentType<any>;
  themeBgClass?: string;
  loadingTitle?: string;
  loadingSubtitle?: string;
}

export default function LegacyOrchestrator({
  themeId,
  onBackToHub,
  particlesComponent,
  themeBgClass = "bg-stone-900",
  loadingTitle = "Invitation",
  loadingSubtitle = "Celebrating Love"
}: LegacyOrchestratorProps) {
  const { invitationData } = useInvitationData();
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    if (invitationData) {
      const timer = setTimeout(() => setLoading(false), 1200);
      return () => clearTimeout(timer);
    }
  }, [invitationData]);

  if (!invitationData) return null;

  const isWrapped = invitationData && "category" in invitationData && "config" in invitationData;
  const invitation = isWrapped ? (invitationData as any).config : invitationData;
  const sectionOrder = invitation.sectionOrder || [];

  return (
    <ScrollController>
      <TemplateFrame themeBgClass={themeBgClass} onBackToHub={onBackToHub}>
        {/* Loading overlay screen */}
        <LoadingExperience loading={loading} title={loadingTitle} subtitle={loadingSubtitle} />

        {/* Particles canvas */}
        <BackgroundEffects
          showParticles={invitation.themeSettings?.showParticles}
          particlesComponent={particlesComponent}
        />

        {/* Audio background playback */}
        <AudioOrchestrator
          musicFile={invitation.music?.file}
          musicEnabled={invitation.music?.enabled}
        />

        {/* Share buttons */}
        <ShareController />

        {/* Sections pipeline render block */}
        <SectionPipeline
          sectionOrder={sectionOrder}
          invitation={invitation}
          themeId={themeId}
        />
      </TemplateFrame>
    </ScrollController>
  );
}
export { default as AudioOrchestrator } from "./AudioOrchestrator";
export { default as BackgroundEffects } from "./BackgroundEffects";
export { default as LoadingExperience } from "./LoadingExperience";
export { default as SectionPipeline } from "./SectionPipeline";
export { default as TemplateFrame } from "./TemplateFrame";
export { default as ScrollController } from "./ScrollController";
export { default as ShareController } from "./ShareController";
