use client";

import React from "react";
import DeitySymbol from "@/components/shared/DeitySymbol";
import LayoutWrapper from "@/components/layouts/LayoutWrapper";

interface HeroSectionProps {
  properties: {
    showScrollIndicator?: boolean;
    overlayOpacity?: number;
    contentAlign?: "left" | "center" | "right";
    minHeight?: string;
  };
  bindings: {
    title?: string;
    subtitle?: string;
    welcomeMessage?: string;
    heroImage?: string;
    ctaText?: string;
    surname?: string;
    subText?: string;
    enableHeaderMedia?: boolean;
    headerMediaType?: "god" | "symbol";
    selectedGod?: "ganesha" | "venkateswara" | "lakshmi" | "shiva" | "krishna";
    selectedSymbol?: "om" | "swastik" | "shree" | "kalash" | "nandi";
    headerMediaSize?: number;
    headerMediaSpacing?: number;
    showSurnameHeading?: boolean;
    surnameFont?: "cinzel" | "playfair" | "great-vibes";
    surnameColor?: string;
  };
  themeId?: string;
  layoutId?: string;
}

export default function HeroSection({ properties, bindings, themeId, layoutId }: HeroSectionProps) {
  const {
    showScrollIndicator = true,
    overlayOpacity = 0.45,
    contentAlign = "center",
    minHeight = "85vh",
  } = properties;

  const {
    title = "",
    subtitle = "",
    welcomeMessage = "",
    heroImage = "",
    ctaText = "",
    surname = "",
    subText = "",
    enableHeaderMedia = true,
    headerMediaType = "god",
    selectedGod = "ganesha",
    selectedSymbol = "om",
    headerMediaSize = 80,
    headerMediaSpacing = 30,
    showSurnameHeading = true,
    surnameFont = "cinzel",
    surnameColor = "var(--theme-accent)",
  } = bindings;

  const alignmentClass = 
    contentAlign === "left" ? "text-left items-start" : 
    contentAlign === "right" ? "text-right items-end" : 
    "text-center items-center";

  return (
    <section 
      className="relative flex flex-col justify-center items-center overflow-hidden w-full select-none"
      style={{ 
        minHeight: "var(--layout-min-height, 85vh)",
        background: "var(--theme-gradient-backdrop, var(--theme-bg))"
      }}
    >
      {/* Background Cover Image */}
      {heroImage && (
        <div 
          className="absolute inset-0 w-full h-full bg-cover bg-center transition-transform duration-700 hover:scale-102"
          style={{ backgroundImage: `url(${heroImage})` }}
        />
      )}

      {/* Dark overlay */}
      <div 
        className="absolute inset-0 bg-black pointer-events-none"
        style={{ opacity: overlayOpacity }}
      />

      {/* Hero content */}
      <LayoutWrapper
        layoutId={layoutId}
        defaultLayoutId="hero-center"
        className={`relative z-10 w-full max-w-4xl flex ${alignmentClass}`}
        style={{ 
          color: "var(--theme-text)",
          fontFamily: "var(--theme-font-body)"
        }}
      >
        {/* Left Side: Welcome, Title, Description, CTA */}
        <div className="flex flex-col space-y-4 flex-1 text-center lg:text-left items-center lg:items-start w-full">
          {/* Welcome Tagline */}
          {welcomeMessage && (
            <p 
              className="text-xs uppercase tracking-[3px] font-semibold"
              style={{ color: "var(--theme-accent)"     }}
            >
              {welcomeMessage}
            </p>
          )}

          {/* Main Title Heading */}
          {title && (
            <h1 
              className="text-4xl md:text-6xl font-extrabold tracking-wide leading-tight"
              style={{ fontFamily: "var(--theme-font-heading)" }}
            >
              {title}
            </h1>
          )}

          {/* Subtitle / Details description */}
          {subtitle && (
            <p 
              className="text-sm max-w-lg leading-relaxed whitespace-pre-line font-medium"
              style={{ color: "var(--theme-secondary-text)" }}
            >
              {subtitle}
            </p>
          )}

          {/* Call to Action Button */}
          {ctaText && (
            <div className="pt-2">
              <button 
                type="button"
                className="px-8 py-3.5 text-xs font-bold uppercase tracking-[2px] transition-all duration-300 shadow-md hover:shadow-lg hover:scale-105 active:scale-95 cursor-pointer"
                style={{
                  backgroundColor: "var(--theme-accent)",
                  color: "#FFFFFF",
                  borderRadius: "var(--theme-radius-button)",
                  boxShadow: "var(--theme-shadow-button)",
                  fontFamily: "var(--theme-font-heading)"
                }}
              >
                {ctaText}
              </button>
            </div>
          )}
        </div>

        {/* Right Side: Deity Icon & Family Surname banner */}
        {(enableHeaderMedia || (showSurnameHeading && surname)) && (
          <div className="flex flex-col items-center justify-center space-y-5 flex-1 w-full pt-6 lg:pt-0">
            {enableHeaderMedia && (
              <div 
                style={{ 
                  marginTop: `${headerMediaSpacing}px`,
                  width: `${headerMediaSize}px`,
                  height: `${headerMediaSize}px` 
                }}
                className="flex items-center justify-center shrink-0"
              >
                <DeitySymbol
                  type={headerMediaType}
                  selectedGod={selectedGod}
                  selectedSymbol={selectedSymbol}
                  size={headerMediaSize}
                />
              </div>
            )}

            {showSurnameHeading && surname && (
              <div className="text-center animate-fadeIn">
                <h2 
                  className="text-2xl md:text-3xl font-bold tracking-wider"
                  style={{ 
                    fontFamily: surnameFont === "great-vibes" ? "var(--font-great)" : surnameFont === "playfair" ? "var(--font-playfair)" : "var(--font-cinzel)",
                    color: surnameColor 
                  }}
                >
                  {surname}
                </h2>
                {subText && (
                  <span className="text-[10px] uppercase tracking-[2px] opacity-75 block mt-1" style={{ color: "var(--theme-secondary-text)" }}>
                    {subText}
                  </span>
                )}
              </div>
            )}
          </div>
        )}
      </LayoutWrapper>

      {/* Scroll indicator */}
      {showScrollIndicator && (
        <div className="absolute bottom-6 left-1/2 -translate-x-1/2 flex flex-col items-center gap-1 z-10 animate-bounce opacity-85">
          <span className="text-[8px] uppercase tracking-[1.5px] text-white/70 font-semibold">Scroll</span>
          <div className="w-1 h-3 bg-white/60 rounded-full" />
        </div>
      )}
    </section>
  );
}
